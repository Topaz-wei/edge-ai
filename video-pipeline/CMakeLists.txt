cmake_minimum_required(VERSION 3.18)
project(edge_video LANGUAGES CXX)
find_package(OpenCV REQUIRED COMPONENTS core imgproc videoio highgui)
add_library(edge_video_config STATIC src/config.cpp)
target_compile_features(edge_video_config PUBLIC cxx_std_17)
target_include_directories(edge_video_config PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(edge_video_config PUBLIC opencv_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(test_video_config tests/test_config.cpp)
    target_link_libraries(test_video_config PRIVATE edge_video_config)
    add_test(NAME video_config COMMAND test_video_config)
endif()
add_library(edge_video_pipeline STATIC src/model.cpp src/pipeline.cpp src/render.cpp)
target_link_libraries(edge_video_pipeline PUBLIC edge_video_config opencv_imgproc opencv_videoio opencv_highgui)
if(BUILD_TESTING)
    add_executable(test_video_pipeline tests/test_pipeline.cpp)
    target_link_libraries(test_video_pipeline PRIVATE edge_video_pipeline)
    add_test(NAME video_pipeline COMMAND test_video_pipeline)
endif()
# Optional model adapters can audit artifacts without putting CUDA/TRT into the video core.
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(edge_engine_artifact STATIC src/engine_artifact.cpp)
target_link_libraries(edge_engine_artifact PUBLIC edge_video_config PRIVATE OpenSSL::Crypto)
if(BUILD_TESTING)
    add_executable(test_engine_artifact tests/test_engine_artifact.cpp)
    target_link_libraries(test_engine_artifact PRIVATE edge_engine_artifact)
    add_test(NAME engine_artifact COMMAND test_engine_artifact)
endif()
