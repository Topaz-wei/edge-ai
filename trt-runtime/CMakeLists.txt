cmake_minimum_required(VERSION 3.18)
project(edge_trt_runtime LANGUAGES CXX)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../cmake")
find_package(CUDAToolkit REQUIRED)
find_package(TensorRT REQUIRED)
add_library(edge_trt_runtime STATIC src/runner.cpp src/tensor_io.cpp)
target_compile_features(edge_trt_runtime PUBLIC cxx_std_17)
target_include_directories(edge_trt_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(edge_trt_runtime PUBLIC TensorRT::Runtime CUDA::cudart)
include(CTest)
if(BUILD_TESTING)
    add_executable(test_tensor_info tests/test_tensor_info.cpp)
    target_link_libraries(test_tensor_info PRIVATE edge_trt_runtime)
    add_test(NAME tensor_info COMMAND test_tensor_info)
endif()

# 真实 GPU 测试需要设备访问权限，单独执行，不混入默认 CPU CTest。
if(BUILD_TESTING)
    add_executable(test_dynamic_runner tests/test_dynamic_runner.cpp)
    target_link_libraries(test_dynamic_runner PRIVATE edge_trt_runtime)
endif()

# 构建库独立链接 ONNX parser，推理库不增加此依赖。
option(EDGE_TRT_BUILD_BUILDER "Build shared ONNX engine builder" ON)
if(EDGE_TRT_BUILD_BUILDER)
    find_library(EDGE_TRT_ONNX_PARSER nvonnxparser REQUIRED)
    add_library(edge_trt_builder STATIC src/builder.cpp)
    target_compile_features(edge_trt_builder PUBLIC cxx_std_17)
    target_include_directories(edge_trt_builder PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
    target_link_libraries(edge_trt_builder PUBLIC TensorRT::Runtime CUDA::cudart PRIVATE ${EDGE_TRT_ONNX_PARSER})
    if(BUILD_TESTING)
        add_executable(test_builder_options tests/test_builder_options.cpp)
        target_link_libraries(test_builder_options PRIVATE edge_trt_builder)
        add_test(NAME builder_options COMMAND test_builder_options)
    endif()
endif()
