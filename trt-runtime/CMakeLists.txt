cmake_minimum_required(VERSION 3.18)
project(edge_trt_runtime LANGUAGES CXX)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../cmake")
find_package(CUDAToolkit REQUIRED)
find_package(TensorRT REQUIRED)
add_library(edge_trt_runtime STATIC src/runner.cpp src/tensor_io.cpp)
target_compile_features(edge_trt_runtime PUBLIC cxx_std_17)
target_include_directories(edge_trt_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(edge_trt_runtime PUBLIC TensorRT::Runtime CUDA::cudart)
include(CTest)
if(BUILD_TESTING)
    add_executable(test_tensor_info tests/test_tensor_info.cpp)
    target_link_libraries(test_tensor_info PRIVATE edge_trt_runtime)
    add_test(NAME tensor_info COMMAND test_tensor_info)
endif()
