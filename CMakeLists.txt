cmake_minimum_required(VERSION 3.18)
project(edge_ai LANGUAGES CXX)
include(CTest)
option(EDGE_BUILD_YOLO "Build YOLO tools and adapter" ON)
option(EDGE_BUILD_RESNET "Build ResNet tools" OFF)
option(EDGE_BUILD_VIDEO "Build the config-driven video application" ON)
if(EDGE_BUILD_VIDEO AND NOT EDGE_BUILD_YOLO)
    message(FATAL_ERROR "EDGE_BUILD_VIDEO currently requires EDGE_BUILD_YOLO; no other image adapter is registered")
endif()
add_subdirectory(trt-runtime)
if(EDGE_BUILD_YOLO)
    set(YOLO_BUILD_VIDEO ${EDGE_BUILD_VIDEO} CACHE BOOL "Build video" FORCE)
    if(EDGE_BUILD_VIDEO AND DEFINED YOLO_BUILD_CUDA_PIPELINE AND NOT YOLO_BUILD_CUDA_PIPELINE)
        message(FATAL_ERROR "Video requires YOLO_BUILD_CUDA_PIPELINE")
    endif()
    add_subdirectory(tensorrt-yolo)
endif()
if(EDGE_BUILD_RESNET)
    add_subdirectory(tensorrt-resnet)
endif()

option(EDGE_BUILD_RTMPOSE "Build RTMPose-Hand offline verification" OFF)
if(EDGE_BUILD_RTMPOSE)
    add_subdirectory(tensorrt-rtmpose-hand)
endif()
